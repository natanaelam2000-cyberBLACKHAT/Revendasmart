// Helper: Structured error response with audit context
function errorResponse(
  res: Response,
  statusCode: number,
  errorType: string,
  message: string,
  context?: Record<string, any>
) {
  const timestamp = new Date().toISOString();
  const errorId = Math.random().toString(36).substring(7);

  console.error(`[${timestamp}] ERROR-ID: ${errorId}`, {
    status: statusCode,
    errorType,
    message,
    context,
  });

  return res.status(statusCode).json({
    error: errorType,
    message,
    errorId,
    timestamp,
    ...(context && { context }),
  });
}

// Middleware: verify Firebase ID token and attach uid to request
async function requireAuth(req: Request, res: Response, next: NextFunction) {
  try {
    const authHeader = req.headers.authorization;
    if (!authHeader?.startsWith("Bearer ")) {
      console.warn("[auth] Missing Authorization header for:", req.path);
      return res.status(401).json({ error: "Unauthorized: missing token" });
    }
    const token = authHeader.slice(7);
    const admin = getFirebaseAdmin();
    const decoded = await admin.auth().verifyIdToken(token);
    (req as any).firebaseUid = decoded.uid;
    next();
  } catch (e) {
    console.warn("[auth] Token verification failed:", (e as any)?.message);
    return res.status(401).json({ error: "Unauthorized: invalid token" });
  }
}

// Middleware: verify the authenticated user owns the resource
function requireOwnership(req: Request, res: Response, next: NextFunction) {
  const firebaseUid = (req as any).firebaseUid;
  const { userId } = req.params;

  if (firebaseUid && userId && firebaseUid !== userId) {
    return res.status(403).json({ error: "Forbidden: you can only access your own data" });
  }

  next();
}

// Middleware: verify admin role via Firebase custom claims (primary) + fallback for migration
async function requireAdmin(req: Request, res: Response, next: NextFunction) {
  const firebaseUid = (req as any).firebaseUid;

  const ADMIN_UIDS_LEGACY = new Set([
    "natanaelam2000@gmail.com",
  ]);

  try {
    const admin = getFirebaseAdmin();
    const userRecord = await admin.auth().getUser(firebaseUid);
    const isAdminClaim = userRecord.customClaims?.["admin"] === true;

    if (isAdminClaim) {
      console.log("[requireAdmin] Access granted via custom claim for:", firebaseUid);
      next();
      return;
    }

    if (ADMIN_UIDS_LEGACY.has(userRecord.email || "")) {
      console.warn(
        "[requireAdmin] MIGRATION: Using legacy email check (deprecated) for:",
        userRecord.email,
        "— Set custom claim to remove fallback"
      );
      next();
      return;
    }

    console.warn("[requireAdmin] Access denied for non-admin user:", firebaseUid, "email:", userRecord.email);
    return res.status(403).json({ error: "Forbidden: admin access required" });
  } catch (e) {
    console.error("[requireAdmin] Error checking admin status:", e);
    return res.status(401).json({ error: "Unauthorized: could not verify admin status" });
  }
}

// Rate limit map: tracks grant requests per admin
const grantRateLimitMap = new Map<string, { count: number; resetAt: number }>();
function checkGrantRateLimit(adminUid: string): boolean {
  const now = Date.now();
  const limit = grantRateLimitMap.get(adminUid);

  if (!limit || now > limit.resetAt) {
    grantRateLimitMap.set(adminUid, { count: 1, resetAt: now + 60000 });
    return true;
  }

  if (limit.count >= 5) {
    return false;
  }

  limit.count++;
  return true;
}

/**
 * Update user's planData in Firestore based on subscription status.
 */
async function syncPlanDataFromSubscription(
  uid: string,
  subscriptionId: string,
  subscriptionStatus: SubscriptionStatus,
  paymentStatus?: string,
  nextBillingDate?: string | null
) {
  const admin = getFirebaseAdmin();
  const db = admin.firestore();
  const planRef = db.collection("users").doc(uid).collection("planData").doc("main");

  const isActive = subscriptionStatus === "authorized";

  const update: Record<string, any> = {
    subscriptionId,
    subscriptionStatus,
    paymentStatus: paymentStatus ?? null,
    autoRenew: isActive,
    updatedAt: admin.firestore.FieldValue.serverTimestamp(),
  };

  console.log(
    `[syncPlanDataFromSubscription] Starting uid=${uid} status=${subscriptionStatus} isActive=${isActive}`
  );

  if (isActive) {
    update.currentPlan = "premium";
    update.premiumActive = true;
    update.premiumSource = "subscription";
    update.premiumExpiresAt = null;
    update.premiumStartedAt = admin.firestore.FieldValue.serverTimestamp();
    update.lastPaymentAt = admin.firestore.FieldValue.serverTimestamp();
    update.canceledAt = null;
    update.graceAccessActive = false;
    update.graceAccessSource = null;
    update.graceAccessExpiresAt = null;
    update.graceAccessCooldownUntil = null;

    if (nextBillingDate) {
      update.nextBillingAt = new Date(nextBillingDate);
    }

    console.log(
      `[syncPlanDataFromSubscription] 🎉 Activating PREMIUM currentPlan=${update.currentPlan} premiumActive=${update.premiumActive}`
    );
  } else if (subscriptionStatus === "cancelled" || subscriptionStatus === "paused") {
    update.currentPlan = "free";
    update.premiumActive = false;
    update.premiumSource = null;
    update.canceledAt = admin.firestore.FieldValue.serverTimestamp();
    update.autoRenew = false;

    console.log(
      `[syncPlanDataFromSubscription] 📉 Deactivating subscription status=${subscriptionStatus}`
    );
  } else if (subscriptionStatus === "pending") {
    // IMPORTANTE:
    // NÃO resetar currentPlan/premiumActive/premiumSource aqui.
    // Pending significa apenas aguardando primeiro pagamento.
    // O usuário pode já ter premium manual/admin e isso não pode ser derrubado.
    update.autoRenew = false;

    console.log(
      `[syncPlanDataFromSubscription] ⏳ Subscription pending, preserving existing premium state`
    );
  }

  console.log(`[syncPlanDataFromSubscription] Update payload:`, JSON.stringify(update, null, 2));

  await planRef.set(update, { merge: true });
  console.log(`[syncPlanDataFromSubscription] ✅ SAVED to Firestore users/${uid}/planData/main`);
  console.log(
    `[syncPlanDataFromSubscription] Result: currentPlan=${update.currentPlan} premiumActive=${update.premiumActive} subscriptionStatus=${subscriptionStatus}`
  );
}

export async function getGlobalConfig(): Promise<GlobalConfig> {
  try {
    const admin = getFirebaseAdmin();
    const db = admin.firestore();
    const configRef = db.collection("system").doc("config");
    const configSnap = await configRef.get();

    if (!configSnap.exists) {
      console.log("[getGlobalConfig] No config found, returning defaults");
      return DEFAULT_GLOBAL_CONFIG;
    }

    const data = configSnap.data();
    return {
      premiumOpenAccess: data?.premiumOpenAccess ?? false,
      premiumOpenAccessUntil: data?.premiumOpenAccessUntil?.toDate() ?? null,
      premiumOpenAccessMessage: data?.premiumOpenAccessMessage ?? null,
    };
  } catch (error) {
    console.error("[getGlobalConfig] Error:", error);
    return DEFAULT_GLOBAL_CONFIG;
  }
}

export function usePlanData() {
  const [uid, setUid] = useState<string | null>(null);
  const [planData, setPlanData] = useState<PlanData | null>(null);
  const [globalConfig, setGlobalConfig] = useState<GlobalConfig>(DEFAULT_GLOBAL_CONFIG);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const auth = getFirebaseAuth();
    if (!auth) {
      setLoading(false);
      return;
    }

    const unsubscribe = onAuthStateChanged(auth, (user) => {
      setUid(user?.uid || null);
    });

    return () => unsubscribe();
  }, []);

  useEffect(() => {
    if (!uid) {
      setPlanData(null);
      setGlobalConfig(DEFAULT_GLOBAL_CONFIG);
      setLoading(false);
      return;
    }

    const db = getFirestore();
    const configRef = doc(db, "system", "config");
    const planDocRef = doc(db, "users", uid, "planData", "main");

    setLoading(true);
    setError(null);

    const unsubscribeConfig = onSnapshot(
      configRef,
      (configSnap) => {
        if (configSnap.exists()) {
          const data = configSnap.data();
          setGlobalConfig({
            premiumOpenAccess: data.premiumOpenAccess ?? false,
            premiumOpenAccessUntil: data.premiumOpenAccessUntil?.toDate?.() ?? null,
            premiumOpenAccessMessage: data.premiumOpenAccessMessage ?? null,
          });
        } else {
          setGlobalConfig(DEFAULT_GLOBAL_CONFIG);
        }
      },
      (err) => {
        console.warn("[usePlanData] Could not load global config, using defaults:", err);
        setGlobalConfig(DEFAULT_GLOBAL_CONFIG);
      }
    );

    const unsubscribePlan = onSnapshot(
      planDocRef,
      (planDocSnap) => {
        try {
          if (planDocSnap.exists()) {
            const data = planDocSnap.data();
            setPlanData(mapPlanData(data, uid));
          } else {
            console.warn("[usePlanData] planData/main não encontrado para UID:", uid);
            setPlanData(null);
          }

          setError(null);
        } catch (err) {
          const errorMsg = err instanceof Error ? err.message : String(err);
          console.error("[usePlanData] ❌ Error processing plan data:", err);

          let userMessage = "Falha ao carregar dados do plano";
          if (errorMsg.includes("Missing or insufficient permissions")) {
            userMessage =
              "Permissão negada ao carregar dados do plano. Entre em contato com suporte.";
          } else if (errorMsg.toLowerCase().includes("auth")) {
            userMessage = "Erro de autenticação. Faça login novamente.";
          }

          setError(userMessage);
        } finally {
          setLoading(false);
        }
      },
      (err) => {
        const errorMsg = err instanceof Error ? err.message : String(err);
        console.error("[usePlanData] ❌ Firestore snapshot error:", err);

        let userMessage = "Falha ao carregar dados do plano";
        if (errorMsg.includes("Missing or insufficient permissions")) {
          userMessage =
            "Permissão negada ao carregar dados do plano. Entre em contato com suporte.";
        } else if (errorMsg.toLowerCase().includes("auth")) {
          userMessage = "Erro de autenticação. Faça login novamente.";
        }

        setError(userMessage);
        setLoading(false);
      }
    );

    return () => {
      unsubscribeConfig();
      unsubscribePlan();
    };
  }, [uid]);

  const refresh = useCallback(async () => {
    if (!uid) return;

    setLoading(true);
    setError(null);

    try {
      const db = getFirestore();

      const configRef = doc(db, "system", "config");
      const planDocRef = doc(db, "users", uid, "planData", "main");

      const [configSnap, planSnap] = await Promise.all([getDoc(configRef), getDoc(planDocRef)]);

      if (configSnap.exists()) {
        const data = configSnap.data();
        setGlobalConfig({
          premiumOpenAccess: data.premiumOpenAccess ?? false,
          premiumOpenAccessUntil: data.premiumOpenAccessUntil?.toDate?.() ?? null,
          premiumOpenAccessMessage: data.premiumOpenAccessMessage ?? null,
        });
      } else {
        setGlobalConfig(DEFAULT_GLOBAL_CONFIG);
      }

      if (planSnap.exists()) {
        setPlanData(mapPlanData(planSnap.data(), uid));
      } else {
        console.warn("[usePlanData] refresh: planData/main não encontrado para UID:", uid);
        setPlanData(null);
      }
    } catch (err) {
      const errorMsg = err instanceof Error ? err.message : String(err);
      console.error("[usePlanData] ❌ Error refreshing plan data:", err);

      let userMessage = "Falha ao atualizar dados do plano";
      if (errorMsg.includes("Missing or insufficient permissions")) {
        userMessage =
          "Permissão negada ao carregar dados do plano. Entre em contato com suporte.";
      } else if (errorMsg.toLowerCase().includes("auth")) {
        userMessage = "Erro de autenticação. Faça login novamente.";
      }

      setError(userMessage);
    } finally {
      setLoading(false);
    }
  }, [uid]);

  const activePlan = getEffectivePlan(planData, globalConfig);

  const shareLink = uid
    ? `${typeof window !== "undefined" ? window.location.origin : ""}?ref=${
        planData?.referralCode || generateReferralCode(uid)
      }`
    : null;

  return {
    planData,
    globalConfig,
    activePlan,
    isPremium: activePlan === PLANS.PREMIUM,
    referralCode: planData?.referralCode || (uid ? generateReferralCode(uid) : null),
    shareLink,
    referralCount: planData?.referralCount || 0,
    loading,
    error,
    refresh,
    subscriptionStatus: planData?.subscriptionStatus ?? null,
    nextBillingAt: planData?.nextBillingAt ?? null,
    canceledAt: planData?.canceledAt ?? null,
    autoRenew: planData?.autoRenew ?? false,
  };
}

        // Validate and persist referral to localStorage
        const isValidUid = /^[a-zA-Z0-9_-]{10,}$/.test(referralUid); // Basic UUID-like validation
        if (isValidUid) {
          try {
            localStorage.setItem("rs:referral_source", referralUid);
            logTelemetryEvent("referral_captured", { referralUid, stage: "app_open", result: "success" }).catch(() => {});
          } catch (e) {
            console.warn("Failed to save referral to localStorage:", e);
          }
        } else {
          // Invalid referral format
          logTelemetryEvent("referral_captured", { referralUid, stage: "app_open", result: "invalid" }).catch(() => {});
        }
        
        // Consume the param from URL to prevent re-processing
        const newUrl = window.location.pathname + window.location.hash;
        window.history.replaceState({ path: newUrl }, '', newUrl);
      }
    }
  }, []); // Run once on mount

  // Monitor Firebase Auth state
  useEffect(() => {
    const auth = getFirebaseAuth();
    if (!auth) {
      setAuthState({ uid: null, loading: false });
      return;
    }

    const unsubscribe = onAuthStateChanged(auth, async (user) => {
      setAuthState({ uid: user?.uid || null, loading: false });
      
      // Initialize plan data on first login
      if (user?.uid) {
        try {
          const token = await user.getIdToken();
          await fetch(getApiUrl(`/api/plan/initialize/${user.uid}`), {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'Authorization': `Bearer ${token}`
            },
          }).catch(err => console.log('[App] Plan init fetch error (expected if already exists):', err.message));
        } catch (err) {
          console.log('[App] Plan initialization error:', err);
        }
      }
    });

    return () => unsubscribe();
  }, []);

    if (authState.loading || settingsLoading) return;

      if (settings?.onboarding_completed !== true) {
        setLocation("/onboarding");
      }
    }
  }, [authState, settings, location, setLocation, settingsLoading]);

  if (app !== null || initError !== null) {

  }

  try {

    const apiKey = firebaseConfig.apiKey;
    const maskedApiKey = apiKey

      : "UNDEFINED";

    console.log("[FIREBASE DEBUG] messagingSenderId:", firebaseConfig.messagingSenderId);

    // Initialize Firebase
    app = initializeApp(firebaseConfig);
    authInstance = getAuth(app);

    // Logs pós-init para garantir qual app/projeto está de pé
    console.log("[FIREBASE DEBUG] initialized app name:", app.name);
    // @ts-ignore
    console.log("[FIREBASE DEBUG] initialized app projectId:", app.options?.projectId);
    // @ts-ignore
    console.log("[FIREBASE DEBUG] initialized app authDomain:", app.options?.authDomain);

    // Initialize error logging after Firebase is initialized
    try {
      initializeErrorLogging(app);
    } catch (err) {
      console.warn("[Firebase] Error logging initialization failed, continuing without it");
    }

    // Initialize internal telemetry after Firebase is initialized
    try {
      initializeInternalTelemetry(app);
    } catch (err) {
      console.warn("[Firebase] Internal telemetry initialization failed, continuing without it");
    }

    // Initialize Firebase Analytics official

      initializeFirebaseAnalytics(app);

import admin from "firebase-admin";

export function initializeFirebaseAdmin() {
  if (initialized && firebaseAdmin) return firebaseAdmin;

  const projectId = process.env.FIREBASE_PROJECT_ID;
  const clientEmail = process.env.FIREBASE_CLIENT_EMAIL;
  const privateKey = process.env.FIREBASE_PRIVATE_KEY?.replace(/\\n/g, "\n");

  if (!projectId || !clientEmail || !privateKey) {
    throw new Error("Missing Firebase Admin credentials in environment");
  }

  try {
    if (!admin.apps.length) {
      admin.initializeApp({
        credential: admin.credential.cert({
          projectId,
          clientEmail,
          privateKey,

        projectId,

    }

    firebaseAdmin = admin;
    initialized = true;
    return firebaseAdmin;
  } catch (error) {
    console.error("[firebase-admin-init] Failed to initialize firebase-admin:", error);
    throw error;
  }
}

    throw new Error("Firebase Admin not initialized");
  }
  return firebaseAdmin;
}
